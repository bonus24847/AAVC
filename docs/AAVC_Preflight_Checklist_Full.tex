% ===================== 1 PRE-DEPARTURE =====================
\section{ก่อนออกสนาม — Pre-departure (ที่แล็บ)}
\begin{cl}
\item \textbf{ชาร์จแบต 17000 ให้เต็ม} (25.1 V / 4.18 V ต่อเซลล์). \emph{ถอดแพ็คหลังชาร์จ} — เสียบค้างไว้ $\sim$0.76 A ต่อชั่วโมงจะกินไปเรื่อย ๆ
\item \textbf{โค้ดล่าสุดอยู่บน CM4}: aavc-practice deploy แล้ว, มี \code{sitl/run_mission.sh} (มี \code{INFRA_ONLY}) และ \code{cm4/start_infra.sh}
\item \textbf{Restart AAVC GCS} ก่อนใช้งาน เพื่อโหลดของที่แก้ล่าสุด (servo fix / chip รั้ว$\to$mission geofence / camera auto-start)
\item (ถ้าเพิ่งแก้โค้ด) \code{make test} + \code{make lint} เขียว
\end{cl}

% ===================== 2 GROUND STATION =====================
\section{ตั้งค่า Ground Station}
\begin{cl}
\item เปิดด้วย \textbf{icon PRACTICE} (ไม่ใช่ COMP) — จะรัน \code{clear_state.sh} ล้าง state เก่าให้เองทุกครั้ง, profile = \code{kmutnb_skyfield}
\item เชื่อม \textbf{laptop $\leftrightarrow$ CM4 hotspot}; ยืนยัน CM4 ตอบ (\code{192.168.28.41}, ssh ได้)
\item เชื่อม \textbf{NOMAD $\leftrightarrow$ FC} (telemetry) และ \textbf{ELRS RC $\leftrightarrow$ FC} (นักบินนิรภัย)
\item AAVC GCS bind \code{127.0.0.1} (เครื่องเดียว) — ถูกต้องแล้ว ไม่ต้องเปิดสู่ LAN
\end{cl}

% ===================== 3 HARDWARE BENCH =====================
\section{ตรวจฮาร์ดแวร์ (bench — ถอดใบพัด) \stoptag}
\crit{รันก่อนทุกครั้ง: \code{.venv/bin/python tools/preflight_params.py} — บล็อก \textbf{BOARD ต้อง 15/15 ถูกทั้งหมด}
ถ้ามีตัวใดผิด = อย่าเพิ่งบิน. บล็อก PINNED เป็นข้อมูลเฉย ๆ (commander pin ให้ตอน connect)}
\begin{cl}
\item \textbf{Motor map} \code{PWM_MAIN_FUNC1..6 = 101..106} — {\color{stopred}\bfseries เคย regress เป็น 0 = มอเตอร์ไม่ผูก output บินไม่ได้; เช็คทุกครั้ง อย่าเชื่อค่าเก่า} (อย่าเผลอเลือก airframe ซ้ำใน QGC — จะล้าง map)
\item \textbf{Servo latch} \code{PWM_AUX_FUNC1..4 = 301..304} (AUX 4/1/2/3 = FL/RR/FR/RL). ทดสอบ \code{ACTUATOR_TEST} เปิด/ปิดครบ 4 มุม; latch หนึ่งต้อง \code{PWM_AUX_MAX = 2100}
\item \textbf{Motor spin} (ถ้าไม่มั่นใจ): \code{ACTUATOR_TEST} M1..M6 ทีละตัว — หมุนครบ 6 ลำดับ+ทิศถูก
\item \textbf{Battery gauge chain}: \code{BAT1_CAPACITY = -1} (voltage-only), \code{BAT1_N_CELLS = 6}, \code{BAT1_V_EMPTY} / \code{BAT1_V_CHARGED} ตรงกับแพ็ค 17000
\item \textbf{TFmini-S}: \code{SENS_TFMINI_CFG} ตั้งพอร์ตแล้ว, \code{EKF2_RNG_CTRL = 1}. {\color{gray}บนพื้นอ่านค่าขยะ (ต่ำกว่า 0.4 m + กล้องสูงจากพื้น 3.5 cm) — เชื่อค่าเฉพาะตอนบิน}
\item \textbf{ไม่มี Crash dump บน SD} — ถ้าขึ้น \emph{``Preflight fail: Crash dump present''} = บล็อก arm (ดู §\ref{sec:trouble})
\item \textbf{Camera} เสียบแน่น (Meige OV9281, nadir, hard-mount ไม่มี gimbal); fov 74.2$^\circ$ วัดแล้ว
\end{cl}

% ===================== 4 BATTERY =====================
\section{แบตเตอรี่}
\begin{cl}
\item \textbf{ชาร์จ $\geq$ 90\%} ก่อนออกบิน; เต็ม = 25.1 V
\item ไม่มี current sensor (PM03D ถอดออก) $\Rightarrow$ gauge เป็น \textbf{voltage-only}: จะ \emph{ตกตอนดันคันเร่งแล้วเด้งกลับ} — safety.py จึงรอ 5 วิ ก่อนตัดสิน (ปกติ)
\item ต่ำกว่า threshold: RTH $\sim$30\%, LAND $\sim$20\% (FC failsafe เป็นตาข่าย)
\item {\color{gray}แพ็ค semi-solid หนัก 1.5 kg, AUW พร้อมบิน $\sim$7.2 kg (MTOW $\leq$ 25 kg — เหลือเยอะ)}
\end{cl}

% ===================== 5 FIELD SETUP =====================
\section{ตั้งสนาม — Field Setup}
\begin{cl}
\item \textbf{pad สะอาด} — ไม่มีกล่อง/ของจากรอบก่อนวางบัง ArUco marker (บินซ้ำโดยไม่ restart stack = marker ถูกบัง)
\item วางอากาศยาน \textbf{อยู่ในรั้ว geofence} ตั้งแต่ต้น (ดู chip ``ในรั้ว'')
\item รอ \textbf{GPS 3D fix + HDOP ต่ำ} — ไม่มี RTK จึงหยาบ; ให้ที่โล่ง เห็นฟ้า ห่างอาคาร
\item ยืนยัน \textbf{profile บน CM4 = \code{kmutnb_skyfield}} (เพดาน 10 m) — ไม่ใช่ comp/KMITL
\end{cl}

% ===================== 7 STAGING =====================
\section{Stage ภารกิจ \& ปล่อยบิน (RC-GO)}
\begin{cl}
\item เลือก \textbf{pad id 4 อัน} จากที่กรรมการ assign (ปุ่มโชว์สัญลักษณ์ ArUco คู่กับเลข) แล้ว \textbf{บันทึก assignment}
\item กด \textbf{ปุ่ม Up ขึ้นโดรน} $\to$ หน้าต่าง checklist $\to$ \textbf{สไลด์ยืนยัน} upload $\to$ ดู progress ขึ้นเครื่อง
\item นักบิน \textbf{arm RC} แล้ว \textbf{สลับเข้า OFFBOARD} — GCS จะแจ้งเขียวเมื่อโดรนเข้า OFFBOARD (เว็บไม่ได้ปล่อยเอง)
\item \textbf{Window 20 นาที เริ่มนับที่ GO แรก} (ในการแข่งจริง เริ่มตอนหัวหน้าทีมแจ้ง ``พร้อม'')
\end{cl}

% ===================== 8 IN-FLIGHT =====================
\section{ระหว่างบิน — Monitoring}
\begin{cl}
\item เพดาน: transit ที่ 20 m (comp) / ปฏิบัติตาม profile ซ้อม; ลงต่ำกว่า 10 m เฉพาะช่วง descent ลง pad
\item เฝ้า: geofence (ในรั้ว), แบต\%, transit pass P1$\to$P3 ทั้งไป-กลับ, การ deliver แต่ละ pad, camera feed
\item อย่าปล่อยไข่กลางอากาศ — ระบบปล่อยหลัง touchdown เท่านั้น (ถ้า telemetry อ่านว่าลอย จะ skip)
\end{cl}

% ===================== 9 EMERGENCY =====================
\section{ขั้นตอนฉุกเฉิน — Emergency}
\crit{นักบินนิรภัยยึดคืนได้ทุกวินาที: \textbf{สลับ POSCTL} $\Rightarrow$ companion stand down ทันที (หยุดส่งคำสั่ง) แล้วบินมือ}
\begin{cl}
\item \textbf{FC failsafes} (อัตโนมัติ): geofence breach $\to$ Return; datalink loss $\to$ Return; RC loss $\to$ Return; แบตต่ำ $\to$ Return แล้ว Land; altitude fence 20 m
\item \textbf{no-fly / เพดาน / floor}: companion watchdog เตือน/RTH อิสระอีกชั้น
\item \textbf{ปล่อย servo มือ}: ทำได้เฉพาะ \emph{บนพื้น} (interlock ล็อกระหว่าง mission บิน)
\item เหตุร้ายแรง: \textbf{kill switch บน RC} (นักบินตัดสิน)
\end{cl}

% ===================== 10 POST-FLIGHT =====================
\section{หลังบิน — Post-flight}
\begin{cl}
\item ลงที่ L\&R แล้ว \textbf{disarm}; เก็บอากาศยาน
\item รัน \textbf{\code{tools/verify_flight.py runs/<id>/audit.jsonl}} — ต้อง PASS (fails closed)
\item ตรวจ: delivered กี่/กี่ pad, id ถูก, ไข่แตกไหม, ระยะปล่อย
\item \textbf{ถอดแพ็คแบต} เก็บ; จด log\% ที่เหลือ
\end{cl}
